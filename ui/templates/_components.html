{# Component macro library for UI redesign Phase 1. Provides reusable Jinja macros for common UI elements such as buttons, cards, tables, forms, modals, toasts, pagination, badges, avatars, and progress bars. All macros use Tailwind CSS utility classes and are accessible (ARIA attributes, focus outlines). They can be imported in any template via: {% raw %}{% import "_components.html" as components %}{% endraw %} #} {# --------------------------------------------------------------------- Button macro Usage: {{ components.button('Save', type='submit', variant='primary') }} Variants: primary, secondary, danger, link Size: sm, md, lg (default md) Additional classes can be passed via ``class``. --------------------------------------------------------------------- #} {% macro button(label, type='button', variant='primary', size='md', class='') %} {% set base = 'inline-flex items-center justify-center font-medium rounded focus:outline-none focus-visible:ring-2 focus-visible:ring-offset-2' %} {% set size_map = {'sm': 'px-2.5 py-1.5 text-sm', 'md': 'px-4 py-2 text-base', 'lg': 'px-6 py-3 text-lg'} %} {% set variant_map = { 'primary': 'bg-indigo-600 text-white hover:bg-indigo-700 focus-visible:ring-indigo-500', 'secondary': 'bg-gray-200 text-gray-800 hover:bg-gray-300 focus-visible:ring-gray-500', 'danger': 'bg-red-600 text-white hover:bg-red-700 focus-visible:ring-red-500', 'link': 'text-indigo-600 hover:underline' } %} {% endmacro %} {# --------------------------------------------------------------------- Card macro Usage: {{ components.card(title='My Card', content='Hello') }} Pass arbitrary HTML via ``content`` (already safe) or use ``caller`` block. --------------------------------------------------------------------- #} {% macro card(title='', content='', class='') %}
{% if title %}
{{ title }}
{% endif %}
{% if caller %}{{ caller() }}{% else %}{{ content|safe }}{% endif %}
{% endmacro %} {# --------------------------------------------------------------------- Table macro Usage: {% raw %}{% call components.table(['Name', 'Age']) %} John30 {% endcall %}{% endraw %} --------------------------------------------------------------------- #} {% macro table(headers, class='') %}
{% for h in headers %} {% endfor %} {{ caller() }}
{{ h }}
{% endmacro %} {# --------------------------------------------------------------------- Form field macro (label + input) Usage: {{ components.form_field('Email', type='email', name='email', required=True) }} --------------------------------------------------------------------- #} {% macro form_field(label, type='text', name='', value='', placeholder='', required=False, minlength=None, pattern=None, rows=None, class='') %}
{% if type == 'textarea' %} {% else %} {% endif %}
{% endmacro %} {% macro form_select(label, name, options, value='', required=False, class='') %}
{% endmacro %} {# --------------------------------------------------------------------- Modal macro (Bootstrap modal wrapper) Usage: {% raw %}{% call components.modal('myModal', title='Confirm') %}

Are you sure?

{% endcall %}{% endraw %} --------------------------------------------------------------------- #} {% macro modal(id, title='', size='md', class='') %}
{% if title %}
{{ title }}
{% endif %}
{{ caller() }}
{% endmacro %} {# --------------------------------------------------------------------- Toast macro (Bootstrap toast) Usage: {{ components.toast('msg-id', 'Success', 'Your changes were saved.', 'success') }} --------------------------------------------------------------------- #} {% macro toast(id, title, body, variant='primary', autohide=True, delay=5000) %}
{{ title }} {{ body }}
{% endmacro %} {# --------------------------------------------------------------------- Pagination macro Usage: {{ components.pagination(page, total_pages, url_for('view', page='')) }} --------------------------------------------------------------------- #} {% macro pagination(current, total, base_url) %}
    {% set prev = current - 1 %}
  • «
    • {% for p in range(1, total + 1) %}
  • {{ p }}
    • {% endfor %} {% set nxt = current + 1 %}
  • »
{% endmacro %} {# --------------------------------------------------------------------- Badge macro Usage: {{ components.badge('New', variant='info') }} --------------------------------------------------------------------- #} {% macro badge(label, variant='secondary', class='') %} {{ label }} {% endmacro %} {# --------------------------------------------------------------------- Avatar macro (initials) Usage: {{ components.avatar('AB', size='md') }} --------------------------------------------------------------------- #} {% macro avatar(initials, size='md', class='') %} {% set size_map = {'sm': 'w-6 h-6 text-sm', 'md': 'w-8 h-8 text-base', 'lg': 'w-12 h-12 text-lg'} %}
{{ initials }}
{% endmacro %} {# --------------------------------------------------------------------- Progress bar macro Usage: {{ components.progress_bar(70, label='70%') }} --------------------------------------------------------------------- #} {% macro progress_bar(percent, label='', class='') %}
{{ label or percent ~ '%' }}
{% endmacro %} {# --------------------------------------------------------------------- Form macro – wraps a
